%--------------------------------------------------------------
\begin{frame}{The question}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{C++-backed layer} (\texttt{rcspp.column\_generation})
      \begin{itemize}\small
        \item loop, master problem, pricer in C++
        \item reached from Python through bindings
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Pure-Python layer} (\texttt{rcspp.python\_column\_generation})
      \begin{itemize}\small
        \item the same features, written in Python
        \item no bindings of its own
      \end{itemize}
    \end{column}
  \end{columns}

  \vspace{0.6em}
  Both call the \textbf{same C++ RCSPP search} for pricing and the same LP solvers (HiGHS through
  \texttt{highspy}, Gurobi through \texttt{gurobipy}).

  \vspace{0.8em}
  \begin{block}{Can we keep only the pure-Python layer?}
    Yes, if it is as fast in practice. Short answer from the measurements:
    \textbf{same results, same speed within measurement noise on realistic runs}.
  \end{block}
\end{frame}

%--------------------------------------------------------------
\begin{frame}[fragile]{Same API in both layers}
  The benchmark drivers switch layers by \textbf{the imported module only}:

  \vspace{0.3em}
\begin{lstlisting}
def layer_module(loop):
    if loop == "purepython":
        import rcspp.python_column_generation as cg
    else:
        import rcspp.column_generation as cg
    return cg

cg = layer_module(loop)                     # the only switch
master = cg.MasterProblem(cg.HighspyMasterSolver(model))   # or GurobipyMasterSolver
graph_pricer = cg.GraphPricer(graph, [cg.PricingStage("simple")])
pricer = master.add_pricer(graph_pricer, cg.ConvexityRow())
result = cg.ColumnGeneration(master, params).run()
result.status, result.objective, result.statistics.generated_columns
\end{lstlisting}

  \small One difference: the pure layer's options are immutable, set in the constructor
  (\texttt{ColumnGenerationParams(initialization=...)}) instead of by assignment.
\end{frame}

%--------------------------------------------------------------
\begin{frame}{What was run, and correctness}
  \begin{itemize}
    \item \textbf{108 cases}: 14 Solomon instances (12 to 100 customers)
      $\times$ \{HiGHS, Gurobi\} $\times$ \{exact pricing, tabu then exact\} $\times$ \{single-customer routes,
      big-M start\}
    \item Both layers back to back in each repeat, on one desktop machine
    \item Repeats by run length: 5 below a few seconds, 3 up to a minute, 1--2 for the longest
    \item \textbf{1,204 runs, 0 failed}
    \item[$\Rightarrow$] \textbf{Same LP value, same iterations, same columns in every case}: the two layers take
      exactly the same path; only the time can differ
  \end{itemize}

  \vspace{0.5em}
  \small Not run: exact pricing from the big-M start on the two longest instances (about 12 h; pricing takes over
  99\,\% of the time there, so the layers cannot differ by more than the noise).
\end{frame}

%--------------------------------------------------------------
\begin{frame}{Proposal}
  \begin{enumerate}
    \item \textbf{Accept the pass rule} as set before the runs (5\,\%, or 0.25\,s for short runs)
    \item \textbf{Keep only the pure-Python layer}: speed is no longer an argument for the C++ one
      \begin{itemize}\small
        \item gain: one implementation; no bindings for the loop; fewer builds, less code; users extend pricers
          and master solvers in Python
        \item cost: \textbf{C++ users lose column generation}; we drop the earlier goal ``usable from C++ and
          Python, loop written once in C++''. If C++ users matter, that is the reason to keep it
        \item work: remove the C++ layer and its bindings, rename the pure package to
          \texttt{column\_generation}; rework or close the open C++ pull requests
      \end{itemize}
    \item \textbf{No C++ helper} to speed up adding columns: it would save at most 0.2\,\% (C201)
  \end{enumerate}

  \vspace{0.4em}
  \small Optional, later: two cheap Python speed-ups (fewer checks when adding columns; not resetting the HiGHS
  options on every solve).
\end{frame}

%--------------------------------------------------------------
\begin{frame}{Caveats}
  \begin{itemize}
    \item The longest runs have only 1--2 repeats (up to 30\,min each)
    \item The machine was not fully idle the first night (browser, editor at times up to a full core); runs
      disturbed by other programs are flagged in the raw results
    \item All runs on one machine; timings from another machine are not comparable
    \item Short runs ($<$ 1\,s) do show the pure layer slower in \%, but by at most 0.1\,s
  \end{itemize}
\end{frame}
