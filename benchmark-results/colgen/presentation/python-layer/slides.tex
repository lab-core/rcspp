\begin{frame}{Decisions to take}
  \begin{enumerate}
    \item \textbf{D-139}: the rule, floor 0.25 s (outcome identical with 0.1 s)
    \item \textbf{Q-41 / D-141}: Python only
      \begin{itemize}\small
        \item gain: one implementation, no bindings layer for the loop
        \item cost: C++ users lose the column generation layer (supersedes D-02, D-08, D-15, D-37)
        \item stack: drafts \#61, \#69, \#72, \#76, Python PRs 5--6, \#80 reworked or closed
      \end{itemize}
    \item \textbf{D-140}: no C++ helper for \texttt{add\_columns} (saves $\le$ 0.2\,\% on C201)
  \end{enumerate}
  \vspace{0.5em}
  \small Caveats: long tier 1--2 repeats; desktop not fully idle; exact pricing from big-M on R201/RC201 not run (pricing $>$ 99\,\%).
\end{frame}
